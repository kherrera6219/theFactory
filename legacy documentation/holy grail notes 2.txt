  intent: string (human-readable description of what this does)
  inputs: InputSpec[]
  outputs: OutputSpec[]
  preconditions: Constraint[]
  postconditions: Constraint[]
  side_effects: SideEffect[]

  source_language: string (e.g., "python", "rust")

  equivalence_tests_total: int
}

  type: LogicType
  constraints: Constraint[]
  optional: boolean
  default: Value | null

OutputSpec {

  constraints: Constraint[]
