3.3.2 Task Routing and Scheduling Algorithms
Primary Routing Strategy: Event-Driven Bus Pattern
The Semantic Bus (Redis) serves as the central routing mechanism. Tasks are not assigned directly to agents; instead, agents subscribe to relevant event channels and react autonomously.

def calculate_priority(task: Task, paradigm: str) -> int:
    """
    Returns priority score (1-100, higher = more urgent)
    """
    base_priority = 50
    
    # Urgency modifiers
    if task.is_blocking_fusion:
        base_priority += 30
    if task.has_external_deadline:
        base_priority += 20
    if task.is_security_critical:
        base_priority += 25
    
    # Paradigm-specific modifiers
    if paradigm == "systems" and task.requires_optimization:
        base_priority += 15  # Systems Pod finishes last in pipeline
    if paradigm == "dynamic" and task.is_ui_related:
        base_priority += 10  # Visual verification depends on this
    
    # Resource availability modifier
    pod_load = get_pod_current_load(paradigm)
    if pod_load > 0.8:
        base_priority -= 10  # Reduce priority if Pod is overloaded
    
    return min(100, max(1, base_priority))
Scheduling Algorithm: Weighted Fair Queuing with Deadlines
python
class TaskScheduler:
    """
    Implements Weighted Fair Queuing with EDF (Earliest Deadline First)
    fallback for deadline-critical tasks.
    """
    
    def __init__(self):
        self.queues = {
            "critical": PriorityQueue(),    # Priority 80-100
            "high": PriorityQueue(),        # Priority 60-79
            "normal": PriorityQueue(),      # Priority 40-59
            "low": PriorityQueue(),         # Priority 1-39
        }
        self.weights = {
            "critical": 8,
            "high": 4,
            "normal": 2,
            "low": 1
        }
    
    def enqueue(self, task: Task):
        if task.priority >= 80:
            self.queues["critical"].push(task, -task.deadline)  # EDF within critical
        elif task.priority >= 60:
            self.queues["high"].push(task, -task.priority)
        elif task.priority >= 40:
            self.queues["normal"].push(task, -task.priority)
        else:
            self.queues["low"].push(task, -task.priority)
    
    def dequeue(self) -> Optional[Task]:
        """
        Weighted fair selection across queues.
        Critical tasks with approaching deadlines always win.
        """
        # Check for deadline-critical tasks first
        if not self.queues["critical"].empty():
            top_task = self.queues["critical"].peek()
            if top_task.deadline - current_time() < DEADLINE_THRESHOLD:
                return self.queues["critical"].pop()
        
        # Weighted round-robin across remaining queues
        total_weight = sum(
            self.weights[q] for q in self.queues 
            if not self.queues[q].empty()
        )
        
        if total_weight == 0:
            return None
        
        selection = random.random() * total_weight
        cumulative = 0
        
        for queue_name in ["critical", "high", "normal", "low"]:
            if not self.queues[queue_name].empty():
                cumulative += self.weights[queue_name]
                if selection < cumulative:
                    return self.queues[queue_name].pop()
        
        return None

async def sub_manager_coordinate_specialists(
    cluster: TaskCluster,
    specialists: List[SpecialistAgent]
) -> GroupStandard:
    """
    Fan-out work to all 4 Specialists in parallel,
    then fan-in results for fusion.
    """
    # Fan-out: dispatch to all Specialists concurrently
    extraction_tasks = [
        asyncio.create_task(
            specialist.extract_logicnode(cluster.get_language_specific(spec.language))
        )
        for specialist in specialists
    ]
    
    # Fan-in: wait for all to complete
    logicnodes = await asyncio.gather(*extraction_tasks, return_exceptions=True)
    
    # Handle any failures
    failed_extractions = [
        (i, result) for i, result in enumerate(logicnodes) 
        if isinstance(result, Exception)
    ]
    
    if failed_extractions:
        # Retry failed extractions with exponential backoff
        logicnodes = await retry_failed_extractions(
            failed_extractions, specialists, cluster
        )
    
    # Detect redundancies
    redundant_pairs = detect_redundant_logic(logicnodes)
    
    # Fuse into Group Standard
    group_standard = fuse_logicnodes(logicnodes, redundant_pairs)
    
    return group_standard

async def parallel_audit_verification(
    logicnode: LogicNode,
    knowledge_lake: VectorDB,
    security_agent: SecurityAgent
) -> AuditResult:
    """
    Run verification stages in parallel where possible.
    """
    # Stage 1: Run equivalence tests and standards check in parallel
    equivalence_task = asyncio.create_task(
        run_equivalence_tests(logicnode, num_simulations=1000)
    )
    standards_task = asyncio.create_task(
        cross_reference_standards(logicnode, knowledge_lake)
    )
    security_task = asyncio.create_task(
        security_agent.scan_logicnode(logicnode)
    )
    
    # Wait for all parallel stages
    equivalence_result, standards_result, security_result = await asyncio.gather(
        equivalence_task, standards_task, security_task
    )
    
    # Stage 2: Tolerance check (depends on equivalence result)
    tolerance_passed = check_tolerance(
        equivalence_result, 
        threshold=0.000001  # 0.0001%
    )
    
    # Final certification decision
    if (tolerance_passed and 
        standards_result.compliant and 
        security_result.secure):
        return AuditResult(
            status="CERTIFIED",
            signature=sign_logicnode(logicnode),
            timestamp=current_timestamp()
        )
    else:
        return AuditResult(
            status="REJECTED",
            correction_memo=generate_correction_memo(
                equivalence_result, standards_result, security_result
            )
        )

class SemanticBusParallelHandler:
    """
    Handles parallel message processing from the Semantic Bus.
    """
    
    def __init__(self, redis_client: Redis, agent_id: str):
        self.redis = redis_client
        self.agent_id = agent_id
        self.subscriptions = []
        self.message_handlers = {}
    
    async def subscribe(self, channels: List[str]):
        """Subscribe to multiple channels for parallel processing."""
        pubsub = self.redis.pubsub()
        for channel in channels:
            await pubsub.subscribe(channel)
            self.subscriptions.append(channel)
        
        # Start parallel message handler
        asyncio.create_task(self._process_messages(pubsub))
    
    async def _process_messages(self, pubsub):
        """Process incoming messages in parallel using a worker pool."""
        async for message in pubsub.listen():
            if message['type'] == 'message':
                # Don't block - spawn handler as separate task
                asyncio.create_task(
                    self._handle_message(message['channel'], message['data'])
                )
    
    async def _handle_message(self, channel: str, data: bytes):
        """Handle individual message without blocking other processing."""
        parsed = json.loads(data)
        
        if channel in self.message_handlers:
            handler = self.message_handlers[channel]
            try:
                await handler(parsed)
            except Exception as e:
                await self._log_handler_error(channel, parsed, e)

class RefineryCheckpointManager:
    """
    Manages checkpoint persistence for the entire Refinery.
    """
    
    def __init__(self, postgres_url: str, redis_client: Redis):
        self.postgres_saver = PostgresSaver.from_conn_string(postgres_url)
        self.redis = redis_client
        self.checkpoint_interval = 30  # seconds
    
    def build_checkpointed_graph(self, workflow: StateGraph) -> CompiledGraph:
        """
        Compile workflow with automatic checkpointing enabled.
        """
        return workflow.compile(checkpointer=self.postgres_saver)
    
    async def checkpoint_agent_state(
        self, 
        agent_id: str, 
        state: AgentState
    ):
        """
        Save agent state to Redis for quick recovery.
        """
        checkpoint_key = f"checkpoint:agent:{agent_id}"
        checkpoint_data = {
            "state": state.to_dict(),
            "timestamp": current_timestamp(),
            "version": state.version
        }
        
        # Set with expiration (keep last 24 hours of checkpoints)
        await self.redis.setex(
            checkpoint_key,
            86400,  # 24 hours
            json.dumps(checkpoint_data)
        )
        
        # Also maintain a rolling history for deeper recovery
        history_key = f"checkpoint:history:{agent_id}"
        await self.redis.lpush(history_key, json.dumps(checkpoint_data))
        await self.redis.ltrim(history_key, 0, 99)  # Keep last 100 checkpoints
    
    async def checkpoint_logicnode(
        self, 
        logicnode: LogicNode
    ):
        """
        Save LogicNode to registry with Git commit.
        """
        # Save to Redis for fast access
        redis_key = f"logicnode:{logicnode.id}"
        await self.redis.set(redis_key, logicnode.to_json())
        
        # Commit to Git for history
        git_commit_logicnode(logicnode)
    
    async def get_global_checkpoint(self) -> GlobalCheckpoint:
        """
        Retrieve the current global state from Postgres.
        """
        # LangGraph handles this through its checkpointer
        latest_checkpoint = await self.postgres_saver.get_latest()
        return GlobalCheckpoint.from_langgraph(latest_checkpoint)

Automatic Recovery on Restart
python
class RefineryRecoveryManager:
    """
    Handles system recovery after crash or restart.
    """
    
    def __init__(
        self,
        checkpoint_manager: RefineryCheckpointManager,
        semantic_bus: SemanticBus,
        agent_registry: AgentRegistry
    ):
        self.checkpoints = checkpoint_manager
        self.bus = semantic_bus
        self.agents = agent_registry
    
    async def recover(self) -> RecoveryReport:
        """
        Full system recovery procedure.
        """
        report = RecoveryReport()
        
        # Step 1: Load global state from Postgres
        global_state = await self.checkpoints.get_global_checkpoint()
        report.global_state_recovered = True
        report.last_checkpoint_time = global_state.timestamp
        
        # Step 2: Identify incomplete work
        incomplete_assignments = self._identify_incomplete_work(global_state)
        report.incomplete_assignments = len(incomplete_assignments)
        
        # Step 3: Recover agent states from Redis
        for agent_id in self.agents.all_ids():
            agent_checkpoint = await self._recover_agent_state(agent_id)
            if agent_checkpoint:
                await self.agents.get(agent_id).restore_state(agent_checkpoint)
                report.agents_recovered.append(agent_id)
        
        # Step 4: Recover verified LogicNodes from Registry
        verified_nodes = await self._recover_logicnodes()
        report.logicnodes_recovered = len(verified_nodes)
        
        # Step 5: Replay incomplete work
        for assignment in incomplete_assignments:
            await self._requeue_assignment(assignment)
            report.assignments_requeued.append(assignment.id)
        
        # Step 6: Validate Traceability Ledger integrity
        ledger_valid = await self._validate_traceability_ledger()
        report.ledger_integrity = ledger_valid
        
        # Step 7: Resume Semantic Bus subscriptions
        await self._restore_bus_subscriptions()
        
        return report
    
    def _identify_incomplete_work(
        self, 
        global_state: GlobalCheckpoint
    ) -> List[Assignment]:
        """
        Find assignments that were in progress when system stopped.
        """
        incomplete = []
        
        for assignment in global_state.active_assignments:
            if assignment.status in ["IN_PROGRESS", "SUBMITTED"]:
                # Check if corresponding LogicNode exists in Registry
                logicnode = self.checkpoints.redis.get(
                    f"logicnode:{assignment.expected_logicnode_id}"
                )
                if not logicnode:
                    incomplete.append(assignment)
        
        return incomplete
    
    async def _requeue_assignment(self, assignment: Assignment):
        """
        Put incomplete assignment back on the Semantic Bus.
        """
        await self.bus.publish(
            channel="/protocol/alpha",
            message=AlphaProtocolMessage(
                type="ASSIGNMENT_RECOVERY",
                source="RECOVERY_MANAGER",
                target=assignment.pod,
                payload=assignment,
                is_recovery=True
            )
        )
Checkpoint Triggers
python
CHECKPOINT_TRIGGERS = {
    # Event-based triggers
    "logicnode_created": True,       # After every LogicNode creation
    "logicnode_verified": True,      # After every Audit certification
    "group_standard_fused": True,    # After every Group fusion
    "pod_complete": True,            # When a Pod finishes all work
    "grand_fusion_complete": True,   # After CEO fusion
    
    # Time-based triggers
    "interval_seconds": 30,          # Periodic checkpoint every 30s
    
    # State-based triggers
    "assignment_count_threshold": 5, # Every 5 assignments
    "logicnode_count_threshold": 10, # Every 10 LogicNodes
}

async def should_checkpoint(event: Event, state: GlobalState) -> bool:
    """Determine if a checkpoint should be taken."""
    
    # Event-based
    if event.type in CHECKPOINT_TRIGGERS and CHECKPOINT_TRIGGERS[event.type]:
        return True
    
    # Time-based
    if (current_time() - state.last_checkpoint_time > 
        CHECKPOINT_TRIGGERS["interval_seconds"]):
        return True
    
    # State-based
    if (state.pending_assignments >= 
        CHECKPOINT_TRIGGERS["assignment_count_threshold"]):
        return True
    
    return False
Rollback Procedure
python
async def rollback_to_checkpoint(
    target_timestamp: datetime,
    checkpoint_manager: RefineryCheckpointManager
) -> RollbackResult:
    """
    Roll back the entire system to a previous checkpoint.
    """
    result = RollbackResult()
    
    # Step 1: Pause all agent activity
    await pause_all_agents()
    
    # Step 2: Find checkpoint at or before target timestamp
    checkpoint = await checkpoint_manager.find_checkpoint_before(target_timestamp)
    if not checkpoint:
        raise CheckpointNotFoundError(f"No checkpoint found before {target_timestamp}")
    
    result.target_checkpoint = checkpoint.id
    result.actual_timestamp = checkpoint.timestamp
    
    # Step 3: Restore Global State Graph
    await checkpoint_manager.postgres_saver.restore(checkpoint.id)
    result.global_state_restored = True
    
    # Step 4: Restore LogicNode Registry to checkpoint state
    # Git handles this via checkout to commit at checkpoint time
    await git_checkout_to_timestamp(checkpoint.timestamp)
    result.registry_restored = True
    
    # Step 5: Clear Redis agent states newer than checkpoint
    await checkpoint_manager.redis.flush_newer_than(checkpoint.timestamp)
    
    # Step 6: Restore agent states from checkpoint
    for agent_id in get_all_agent_ids():
        agent_checkpoint = await checkpoint_manager.get_agent_checkpoint(
            agent_id, checkpoint.timestamp
        )
        if agent_checkpoint:
            await restore_agent_state(agent_id, agent_checkpoint)
    
    # Step 7: Resume agent activity
    await resume_all_agents()
    
    result.success = True
    return result
